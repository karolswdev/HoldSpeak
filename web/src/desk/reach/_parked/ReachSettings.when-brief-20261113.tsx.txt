                            value={status?.channels.phone_app.on === false ? "off" : "on"}
                            options={PHONE_RUNG}

                          />

                      kind: "preference",
                      name: entry.label,
                    })),
                  ]}

                  columns={[{ key: "name", word: "Name" }]}

                  onSelect={pickBrief}

                />
