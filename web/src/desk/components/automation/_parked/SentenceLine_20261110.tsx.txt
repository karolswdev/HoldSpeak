// Parked: the object uses shared AutomationSetup and kit reading/list panes.
/* One pressable part of the mast sentence. The library popover owns focus,
 * Escape and return focus; setup uses these same parts with empty values. */
import { type ReactNode, type RefObject } from "react";
import { Button } from "../../../components/signal/Signal";
import { Popover } from "../../surface";

export function SentenceLine({ anchor, name, label, children, open = false, onPress, editor, blank = false }: {
  anchor: RefObject<HTMLElement | null>; name: string; label: string; children: ReactNode; open?: boolean;
  onPress?(): void; editor?: ReactNode; blank?: boolean; variant?: "line" | "lead" | "fact";
}) {
  const join = ({ about: " about ", with: " with ", put: " into ", review: ", reviewed by " } as Record<string, string>)[name] ?? "";
  return <span className="automation-line" data-line={name} data-blank={blank || undefined}
    data-open={open || undefined} data-testid={`automation-line-${name}`}>
    {join ? <span className="automation-join">{join}</span> : null}
    {onPress ? <Button variant="chrome" className="automation-line-press" aria-label={label} aria-expanded={open}
      onClick={onPress}>{children}</Button> : <span>{children}</span>}
    <Popover anchor={anchor} open={open && Boolean(editor)} onClose={() => onPress?.()} ariaLabel={`Change ${label.toLowerCase()}`}>
      <div className="automation-part-editor">{editor}</div>
    </Popover>
  </span>;
}
