class _Text(HTMLParser):
    def __init__(self) -> None:
        super().__init__(convert_charrefs=True)
        self.parts: list[str] = []
        self.title: list[str] = []
        self.in_title = False

    def handle_starttag(self, tag: str, attrs: list[tuple[str, str | None]]) -> None:
        self.in_title = tag.lower() == "title"

    def handle_endtag(self, tag: str) -> None:
        if tag.lower() == "title":
            self.in_title = False

    def handle_data(self, data: str) -> None:
        if self.in_title:
            self.title.append(data)
        self.parts.append(data)


def _clean(chunks: Iterable[str]) -> str:
    return re.sub(r"\s+", " ", " ".join(chunks)).strip()


def _home_isolated() -> bool:
    """Require HOME to differ from the invoking user's real home directory."""
    configured = os.environ.get("HOME", "").strip()
    if not configured:
        return False
    try:
        real_home = pwd.getpwuid(os.getuid()).pw_dir
    except (KeyError, OSError):
        return False
    return Path(configured).expanduser().resolve() != Path(real_home).expanduser().resolve()


def _fetch(label: str, url: str) -> tuple[str, int, str, str]:
    parsed = urlparse(url)
    if parsed.scheme != "https" or parsed.hostname not in ALLOWED_HOSTS:
        raise RuntimeError(f"refusing non-official URL: {url}")
    request = Request(url, method="GET", headers={"User-Agent": "HoldSpeak-Docs-Probe/1"})
    with urlopen(request, timeout=TIMEOUT) as response:  # noqa: S310 - fixed official GET URL
        final = response.geturl()
        final_host = urlparse(final).hostname
        if final_host not in ALLOWED_HOSTS:
            raise RuntimeError(f"redirect escaped official docs: {final}")
        raw = response.read(MAX_BYTES + 1)
        if len(raw) > MAX_BYTES:
            raise RuntimeError(f"page too large: {label}")
        parser = _Text()
        parser.feed(raw.decode("utf-8", errors="replace"))
        return final, int(response.status), _clean(parser.title), _clean(parser.parts).lower()
