    35	const isPanelId = (value: unknown): value is string =>
    36	  typeof value === "string" && /^[A-Za-z0-9:_-]+$/.test(value);
    37
    38	const isPanelRect = (value: unknown): value is PanelRect => {
    39	  if (!value || typeof value !== "object") return false;
    40	  const rect = value as PanelRect;
    41	  return [rect.x, rect.y, rect.w, rect.h].every(Number.isFinite) &&
    42	    rect.w > 0 && rect.h > 0;
    43	};

    45	const compactIds = (values: unknown, limit = PANEL_ORDER_LIMIT): string[] => {
    46	  if (!Array.isArray(values)) return [];
    47	  const unique = Array.from(new Set(values.filter(isPanelId)));
    48	  return unique.length > limit ? unique.slice(-limit) : unique;
    49	};
    50
    51	const isZoneViewPref = (value: unknown): value is ZoneViewPref => {
    52	  if (!value || typeof value !== "object") return false;
    53	  const pref = value as ZoneViewPref;
    54	  return (
    55	    (pref.view === "icons" || pref.view === "list") &&
    56	    (pref.sort === "name" || pref.sort === "kind" || pref.sort === "modified") &&
    57	    (pref.dir === "asc" || pref.dir === "desc")
    58	  );

    61	function parseWindows(value: unknown): Record<string, WindowInstance> {
    62	  if (!value || typeof value !== "object") return {};
    63	  const windows: Record<string, WindowInstance> = {};
    64	  for (const [id, raw] of Object.entries(value)) {
    65	    if (!isPanelId(id) || !raw || typeof raw !== "object") continue;
    66	    const candidate = raw as Partial<WindowInstance>;
    67	    if (
    68	      candidate.id !== id ||
    69	      candidate.kind !== "surface" ||
    70	      typeof candidate.applicationKey !== "string" ||
    71	      (candidate.scope !== null && typeof candidate.scope !== "string") ||
    72	      candidate.persistence !== "workspace"
    73	    ) continue;
    74	    windows[id] = candidate as WindowInstance;
    75	  }
    76	  return windows;
    77	}

    30	/** Build the open method for one window type. */
    31	export function makeOpenWindow<K extends string>(
    32	  config: WindowTypeConfig<K>,
    33	) {
    34	  return function openWindow(
    35	    this: void,
    36	    value: string,
    37	    origin: { x: number; y: number } | undefined,
    38	    set: (partial: Partial<DeskState>) => void,
    39	    get: () => DeskState,
    40	  ): void {
    41	    const open = get()[config.field] as unknown as WindowEntry<K>[];
    42	    const panelId = `${config.panelPrefix}${value}`;
    43	    if (!open.some((w) => w[config.key] === value)) {
    44	      const entry = { [config.key]: value, origin: origin ?? null } as unknown as WindowEntry<K>;
    45	      const next = [...open, entry];
    46	      set({ [config.field]: next } as unknown as Partial<DeskState>);
    47	      if (config.onOpen) config.onOpen(value, get, set);
    48	      sfx("latch");
    49	    }
    50	    get().focusPanel(panelId);
    51	  };
    52	}

    32	/** HS-97-02 — the open-placement engine. A window opening without a
    33	 * persisted rect lands FULLY inside the working band (below the chrome,
    34	 * clear of the dock), seeded at its CSS default home but moved off other
    35	 * windows' title bars by a min-overlap scan (head occlusion dominates,
    36	 * then overlap area, then distance from home). Pure, pinned by test. */
    37	export function placeWindow(
    38	  seed: PanelRect,
    39	  existing: PanelRect[],
    40	  vw: number,
    41	  vh: number,
    42	  minW = 320,
    43	  minH = 220,
    44	): PanelRect {
    45	  const { top, bottom } = workBand();
    46	  const w = Math.max(minW, Math.min(seed.w, vw - MARGIN * 2));
    47	  const h = Math.max(minH, Math.min(seed.h, Math.max(minH, vh - top - bottom)));
    48	  const maxX = Math.max(MARGIN, vw - MARGIN - w);
    49	  const maxY = Math.max(top, vh - bottom - h);
    50	  const sx = Math.min(Math.max(seed.x, MARGIN), maxX);
    51	  const sy = Math.min(Math.max(seed.y, top), maxY);
    52	  const overlap = (
    53	    ax: number,
    54	    ay: number,
    55	    aw: number,
    56	    ah: number,
    57	    b: PanelRect,
    58	    bh: number,
    59	  ) => {
    60	    const ox = Math.max(0, Math.min(ax + aw, b.x + b.w) - Math.max(ax, b.x));
    61	    const oy = Math.max(0, Math.min(ay + ah, b.y + bh) - Math.max(ay, b.y));
    62	    return ox * oy;
    63	  };
    64	  const score = (x: number, y: number) => {
    65	    let heads = 0;
    66	    let area = 0;
    67	    for (const r of existing) {
    68	      area += overlap(x, y, w, h, r, r.h);
    69	      if (overlap(x, y, w, HEAD, r, HEAD) > 0) heads++;
    70	    }
    71	    return heads * 1e9 + area * 10 + Math.hypot(x - sx, y - sy);
    72	  };
    73	  let best = { x: sx, y: sy, s: score(sx, sy) };
    74	  const STEP = 32;
    75	  for (let y = top; y <= maxY; y += STEP) {
    76	    for (let x = MARGIN; x <= maxX; x += STEP) {
    77	      const s = score(x, y);
    78	      if (s < best.s - 0.5) best = { x, y, s };
    79	    }
    80	  }
    81	  if (best.s >= 1e9) {
    82	    // Saturated: every position occludes a title bar somewhere. The
    83	    // cascade survives exactly here — step down-right off the home seat.
    84	    const step = CASCADE * Math.min(existing.length, 8);
    85	    return {
    86	      x: Math.min(Math.max(sx + step, MARGIN), maxX),
    87	      y: Math.min(Math.max(sy + step, top), maxY),
    88	      w,
    89	      h,
    90	    };
    91	  }
    92	  return { x: best.x, y: best.y, w, h };
    93	}

   113	/** HS-95-03 — edge snap: releasing a window drag at a screen edge tiles
   114	 * it. Corners take quarters, the left/right flanks take halves; anywhere
   115	 * else returns null (a free park). Pure, pinned by test. */
   116	export function snapForPointer(
   117	  px: number,
   118	  py: number,
   119	  vw: number,
   120	  vh: number,
   121	): PanelRect | null {
   122	  const EDGE = 26;
   123	  const CORNER = 150;
   124	  const { top, bottom } = workBand(); // below chrome, clear of dock
   125	  const halfW = Math.floor((vw - MARGIN * 3) / 2);
   126	  const halfH = Math.floor((vh - top - bottom - MARGIN) / 2);
   127	  const left = px <= CORNER;
   128	  const right = px >= vw - CORNER;
   129	  const high = py <= CORNER + top;
   130	  const low = py >= vh - CORNER;
   131	  if (left && high) return { x: MARGIN, y: top, w: halfW, h: halfH };
   132	  if (right && high)
   133	    return { x: vw - MARGIN - halfW, y: top, w: halfW, h: halfH };
   134	  if (left && low)
   135	    return { x: MARGIN, y: top + halfH + MARGIN, w: halfW, h: halfH };
   136	  if (right && low)
   137	    return {
   138	      x: vw - MARGIN - halfW,
   139	      y: top + halfH + MARGIN,
   140	      w: halfW,
   141	      h: halfH,
   142	    };
   143	  if (px <= EDGE)
   144	    return { x: MARGIN, y: top, w: halfW, h: vh - top - bottom };
   145	  if (px >= vw - EDGE)
   146	    return { x: vw - MARGIN - halfW, y: top, w: halfW, h: vh - top - bottom };
   147	  return null;
   148	}
   186	/** HS-97-06 — the expose grid: N non-overlapping cells inside the
   187	 * working band, last row centered. Pure, pinned by test. */
   188	export function exposeLayout(
   189	  count: number,
   190	  vw: number,
   191	  vh: number,
   192	): PanelRect[] {
   193	  const band = workBand();
   194	  const top = band.top + 8;
   195	  const bottom = band.bottom + 8;
   196	  const GAP = 18;
   197	  const cols = Math.max(1, Math.ceil(Math.sqrt(count)));
   198	  const rows = Math.max(1, Math.ceil(count / cols));
   199	  const bandW = vw - MARGIN * 2;
   200	  const bandH = vh - top - bottom;
   201	  const w = Math.floor((bandW - GAP * (cols - 1)) / cols);
   202	  const h = Math.floor((bandH - GAP * (rows - 1)) / rows);
   203	  const cells: PanelRect[] = [];
   204	  for (let i = 0; i < count; i++) {
   205	    const r = Math.floor(i / cols);
   206	    const inRow = r === rows - 1 ? count - r * cols : cols;
   207	    const rowW = inRow * w + (inRow - 1) * GAP;
   208	    const x0 = MARGIN + Math.floor((bandW - rowW) / 2);
   209	    cells.push({
   210	      x: x0 + (i - r * cols) * (w + GAP),
   211	      y: top + r * (h + GAP),
   212	      w,
   213	      h,
   214	    });
   215	  }
   216	  return cells;
   217	}
