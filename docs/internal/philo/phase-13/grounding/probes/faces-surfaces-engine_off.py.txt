import sys, json
sys.path.insert(0, '.')
from fslib import *
from walk import enter_desk, press, try_open
out = {}
for w in (1440, 393):
    with browser(w) as page:
        enter_desk(page, 'pop')
        try_open(page, 'go:Ask AI', 'pop'); page.wait_for_selector('.desk-window[aria-label="Ask AI"]'); settle(page, 600)
        win = page.locator('.desk-window[aria-label="Ask AI"]')
        import re as _re
        b = win.locator('button').filter(has_text=_re.compile(r'^\W*ASK\s*$')).first
        out[f'ask_label_{w}'] = b.inner_text() if b.count() else 'absent'
        out[f'ask_disabled_{w}'] = b.get_attribute('disabled') if b.count() else 'absent'
        try:
            press(page, b); page.wait_for_timeout(2500); settle(page, 300)
        except Exception as exc:
            out[f'ask_press_err_{w}'] = repr(exc)[:150]
        out[f'ask_after_{w}'] = page.evaluate("() => [...document.querySelectorAll('.desk-window')].map(e => e.getAttribute('aria-label') + ' :: ' + e.innerText.replace(/\\s+/g, ' ').slice(0, 500))")
        out[f'shot_ask_{w}'] = shot(page, f'120-engine-off-ask-pop-{w}')
    with browser(w) as page:
        enter_desk(page, 'pop')
        try_open(page, 'go:Speak', 'pop'); page.wait_for_selector('.desk-window[aria-label="Speak"]'); settle(page, 600)
        win = page.locator('.desk-window[aria-label="Speak"]')
        import re as _re
        t = win.locator('button').filter(has_text=_re.compile(r'TALK')).first
        try:
            press(page, t); page.wait_for_timeout(2500); settle(page, 300)
        except Exception as exc:
            out[f'talk_err_{w}'] = repr(exc)[:150]
        out[f'talk_after_{w}'] = page.evaluate("() => [...document.querySelectorAll('.desk-window')].map(e => e.getAttribute('aria-label') + ' :: ' + e.innerText.replace(/\\s+/g, ' ').slice(0, 500))")
        out[f'shot_talk_{w}'] = shot(page, f'121-engine-off-talk-pop-{w}')
print(json.dumps(out, indent=1)[:5000])
dump('engine-off', out)
