import sys, json
sys.path.insert(0, '.')
from fslib import *
from walk import enter_desk, press, try_open
from pullouts import to_list
out = {}
with browser(393) as page:
    enter_desk(page, 'pop')
    press(page, page.locator('.desk-menubar').get_by_text('Search').first); settle(page, 300)
    page.keyboard.type('People'); settle(page, 500)
    out['palette_people_393'] = page.evaluate("() => { const p = document.querySelector('[role=listbox]') || document.querySelector('[class*=palette], [class*=shelf], [class*=deck]'); return p ? p.innerText.replace(/\\s+/g, ' ').slice(0, 600) : document.body.innerText.slice(0, 300); }")
    out['shot_pal_people_393'] = shot(page, '110-palette-people-pop-393')
    page.keyboard.press('Escape'); settle(page, 300)
    out['shelf_open_after_one_escape'] = page.evaluate("() => { const s = document.getElementById('desk-tool-shelf'); return !!s && s.getBoundingClientRect().height > 0; }")
    page.keyboard.press('Escape'); settle(page, 300)
    out['shelf_open_after_two_escapes'] = page.evaluate("() => { const s = document.getElementById('desk-tool-shelf'); return !!s && s.getBoundingClientRect().height > 0; }")
with browser(1440) as page:
    enter_desk(page, 'pop')
    to_list(page, 1440)
    row = page.locator('tr.desk-sortable-table-row', has_text='Freeze the old ledger on Nov 5').first
    row.evaluate("e => e.scrollIntoView({block: 'center'})")
    row.locator('td').nth(1).click(); settle(page, 300)
    page.locator('nav.desk-verbbar').get_by_text('Object', exact=True).first.click(); settle(page, 300)
    out['object_menu_selected'] = page.evaluate("() => [...document.querySelectorAll('[role=menu] [role^=menuitem]')].map(e => (e.getAttribute('aria-disabled') === 'true' ? '(ghost) ' : '') + e.innerText.replace(/\\s+/g, ' ').trim())")
    out['shot_object_menu'] = shot(page, '111-object-menu-decision-pop-1440')
    gi = page.get_by_role('menuitem', name='Get Info').first
    out['get_info_ghosted'] = gi.get_attribute('aria-disabled')
    if gi.get_attribute('aria-disabled') != 'true':
        gi.click(); settle(page, 1000)
    else:
        page.keyboard.press('Escape'); settle(page, 200)
        row.locator('td').nth(1).click(button='right'); settle(page, 400)
        out['row_context_menu'] = page.evaluate("() => [...document.querySelectorAll('[role=menu] [role^=menuitem]')].map(e => (e.getAttribute('aria-disabled') === 'true' ? '(ghost) ' : '') + e.innerText.replace(/\\s+/g, ' ').trim())")
        out['shot_row_menu'] = shot(page, '111b-row-context-menu-decision-pop-1440')
        it = page.get_by_role('menuitem', name='Get Info')
        if it.count() and it.first.get_attribute('aria-disabled') != 'true':
            it.first.click(); settle(page, 1000)
    out['get_info'] = page.evaluate("() => { const ws = [...document.querySelectorAll('.desk-window')]; const w = ws[ws.length - 1]; return w ? w.getAttribute('aria-label') + ' :: ' + w.innerText.replace(/\\s+/g, ' ').slice(0, 600) : null; }")
    out['shot_get_info'] = shot(page, '112-get-info-decision-pop-1440')
print(json.dumps(out, indent=1))
dump('misc', out)
