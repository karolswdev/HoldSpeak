import sys, json, time
sys.path.insert(0, '.')
from fslib import *
from walk import enter_desk, press, try_open
w = int(sys.argv[1])
out = {}
with browser(w) as page:
    enter_desk(page, 'pop')
    press(page, page.locator('.desk-dock').get_by_role('button', name='Floor').first); settle(page, 1200)
    if w > 600:
        page.locator('nav.desk-verbbar').get_by_text('Desk', exact=True).first.click(); settle(page, 300)
        page.get_by_role('menuitem', name='List view').first.click(); settle(page, 1000)
    page.wait_for_timeout(800); print('rows', page.evaluate("() => [...document.querySelectorAll('[role=row], tr, li')].filter(e => /Payments ledger/.test(e.innerText)).map(e => e.tagName + '.' + e.className).slice(0,5)")); row = page.locator('[role=row], tr, li', has_text='Payments ledger cutover').last
    row.evaluate("e => e.scrollIntoView({block: 'center'})")
    t0 = time.monotonic()
    if w > 600: row.dblclick()
    else: row.tap()
    page.wait_for_timeout(300)
    for _ in range(60):
        if page.evaluate("() => [...document.querySelectorAll('.desk-window')].some(e => /ledger/i.test(e.getAttribute('aria-label') || ''))"): break
        page.wait_for_timeout(100)
    settle(page, 1500)
    out['open_ms'] = round((time.monotonic() - t0) * 1000)
    out['windows'] = page.evaluate("() => [...document.querySelectorAll('.desk-window')].map(e => e.getAttribute('aria-label') + ' :: ' + e.innerText.replace(/\\s+/g,' ').slice(0, 1500))")
    out['shot'] = shot(page, f'64-room-pop-{w}')
    # all verbs in the Room
    out['verbs'] = page.evaluate("() => { const f = [...document.querySelectorAll('.desk-window')].find(e => /ledger/i.test(e.getAttribute('aria-label') || '')); return f ? [...f.querySelectorAll('button')].map(b => (b.getAttribute('aria-label') || b.innerText).trim().replace(/\\s+/g, ' ').slice(0, 40) + (b.disabled ? ' (disabled)' : '')) : null; }")
    page.evaluate("() => { const f = [...document.querySelectorAll('.desk-window')].find(e => /ledger/i.test(e.getAttribute('aria-label') || '')); const t = [...f.querySelectorAll('*')].find(e => e.children.length === 0 && /DECISIONS/.test(e.textContent)); if (t) t.scrollIntoView({block: 'start'}); }")
    settle(page, 500)
    out['shot_decisions'] = shot(page, f'65-room-decisions-pop-{w}')
    out['decision_row'] = page.evaluate("() => { const f = [...document.querySelectorAll('.desk-window')].find(e => /ledger/i.test(e.getAttribute('aria-label') || '')); const t = [...f.querySelectorAll('*')].filter(e => /Freeze the old ledger/.test(e.innerText || '') ).pop(); let r = t; for (let i = 0; i < 4 && r; i++) r = r.parentElement; return r ? {html: r.outerHTML.slice(0, 1500), buttons: [...r.querySelectorAll('button')].map(b => b.innerText.trim())} : null; }")
    dock = page.evaluate("() => { const d = document.querySelector('.desk-dock'); const r = d.getBoundingClientRect(); return {left: Math.round(r.left), right: Math.round(r.right), sw: d.scrollWidth, cw: d.clientWidth, vw: innerWidth}; }")
    out['dock_geometry'] = dock
    dr = page.locator('[data-testid=decision-row]').first
    press(page, dr); settle(page, 800)
    out['decision_unfolded'] = page.evaluate("() => { const r = document.querySelector('[data-testid=decision-row]').closest('li'); return {text: r.innerText.replace(/\\s+/g, ' ').slice(0, 600), buttons: [...r.querySelectorAll('button')].map(b => b.innerText.trim())}; }")
    out['shot_unfolded'] = shot(page, f'66-room-decision-unfolded-pop-{w}')
print(json.dumps(out, indent=1)[:7000])
dump(f'room-{w}', out)
