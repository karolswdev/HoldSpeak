    16	  version: typeof DESK_WORKSPACE_VERSION;
    17	  windowsById: Record<string, WindowInstance>;

    20	    order: string[];

    27	const emptyWorkspace = (): DeskWorkspaceDocumentV1 => ({
    28	  version: DESK_WORKSPACE_VERSION,
    29	  windowsById: {},
    30	  panel: { rects: {}, order: [], max: [] },
    31	  zoneWindows: [],
    32	  zoneViewPrefs: {},

    34
    35	const isPanelId = (value: unknown): value is string =>
    36	  typeof value === "string" && /^[A-Za-z0-9:_-]+$/.test(value);
    37
    38	const isPanelRect = (value: unknown): value is PanelRect => {
    39	  if (!value || typeof value !== "object") return false;
    40	  const rect = value as PanelRect;
    41	  return [rect.x, rect.y, rect.w, rect.h].every(Number.isFinite) &&
    42	    rect.w > 0 && rect.h > 0;
    43	};
    44
    45	const compactIds = (values: unknown, limit = PANEL_ORDER_LIMIT): string[] => {
    46	  if (!Array.isArray(values)) return [];
    47	  const unique = Array.from(new Set(values.filter(isPanelId)));
    48	  return unique.length > limit ? unique.slice(-limit) : unique;
    49	};
    50
    51	const isZoneViewPref = (value: unknown): value is ZoneViewPref => {
    52	  if (!value || typeof value !== "object") return false;
    53	  const pref = value as ZoneViewPref;
    54	  return (
    55	    (pref.view === "icons" || pref.view === "list") &&
    56	    (pref.sort === "name" || pref.sort === "kind" || pref.sort === "modified") &&
    57	    (pref.dir === "asc" || pref.dir === "desc")
    58	  );
    59	};
    60
    61	function parseWindows(value: unknown): Record<string, WindowInstance> {
    62	  if (!value || typeof value !== "object") return {};
    63	  const windows: Record<string, WindowInstance> = {};
    64	  for (const [id, raw] of Object.entries(value)) {
    65	    if (!isPanelId(id) || !raw || typeof raw !== "object") continue;
    66	    const candidate = raw as Partial<WindowInstance>;
    67	    if (
    68	      candidate.id !== id ||
    69	      candidate.kind !== "surface" ||
    70	      typeof candidate.applicationKey !== "string" ||
    71	      (candidate.scope !== null && typeof candidate.scope !== "string") ||
    72	      candidate.persistence !== "workspace"
    73	    ) continue;
    74	    windows[id] = candidate as WindowInstance;
    75	  }
    76	  return windows;
    77	}

    79	export function loadDeskWorkspace(): DeskWorkspaceDocumentV1 {
    80	  try {
    81	    const raw: unknown = JSON.parse(
    82	      localStorage.getItem(DESK_WORKSPACE_STORAGE_KEY) || "null",
    83	    );
    84	    if (!raw || typeof raw !== "object") return emptyWorkspace();
    85	    const candidate = raw as Partial<DeskWorkspaceDocumentV1>;
    86	    if (candidate.version !== DESK_WORKSPACE_VERSION)
    87	      return emptyWorkspace();
    88
    89	    const panel = candidate.panel && typeof candidate.panel === "object"
    90	      ? candidate.panel
    91	      : { rects: {}, order: [], max: [] };
    92	    const rects = Object.fromEntries(
    93	      Object.entries(panel.rects && typeof panel.rects === "object" ? panel.rects : {})
    94	        .filter(([id, rect]) => isPanelId(id) && isPanelRect(rect)),
    95	    ) as Record<string, PanelRect>;
    96	    const zoneViewPrefs = Object.fromEntries(
    97	      Object.entries(
    98	        candidate.zoneViewPrefs && typeof candidate.zoneViewPrefs === "object"
    99	          ? candidate.zoneViewPrefs
   100	          : {},
   101	      ).filter(([id, pref]) => isPanelId(id) && isZoneViewPref(pref)),
   102	    ) as Record<string, ZoneViewPref>;
   103
   104	    return {
   105	      version: DESK_WORKSPACE_VERSION,
   106	      windowsById: parseWindows(candidate.windowsById),
   107	      panel: {
   108	        rects,
   109	        order: compactIds(panel.order),
   110	        max: compactIds(panel.max),
   111	      },
   112	      zoneWindows: compactIds(candidate.zoneWindows),
   113	      zoneViewPrefs,
   114	    };
   115	  } catch {
   116	    return emptyWorkspace();
   117	  }
   118	}

   132	  const rects: Record<string, PanelRect> = {};
   133	  for (const id of state.panelSaved) {
   134	    const rect = state.panelRects[id];
   135	    if (rect && isPanelId(id) && isPanelRect(rect)) rects[id] = rect;
   136	  }

   138	    version: DESK_WORKSPACE_VERSION,
   139	    windowsById: state.windowsById,

   142	      order: compactIds(state.panelOrder),

   144	    },
   145	    zoneWindows: state.zoneWindows.map((window) => window.id),
   146	    zoneViewPrefs: state.zoneViewPrefs,
   147	  };

   149	    localStorage.setItem(DESK_WORKSPACE_STORAGE_KEY, JSON.stringify(document));

   152	  }

    32	    const set = listeners.current.get(type) ?? new Set<Listener>();
    33	    set.add(listener);
    34	    listeners.current.set(type, set);
    35	    return () => {
    36	      set.delete(listener);

    42	    let socket: WebSocket | null = null;
    43	    let timer = 0;
    44	    let ping = 0;
    45	    let attempt = 0;
    46	    let disposed = false;
    47
    48	    const connect = () => {
    49	      if (disposed) return;
    50	      setState(attempt ? "reconnecting" : "connecting");
    51	      socket = new WebSocket(websocketUrl(), websocketProtocols());
    52	      socket.addEventListener("open", () => {
    53	        attempt = 0;
    54	        setState("connected");

    60	        if (typeof event.data !== "string") return;
    61	        try {
    62	          const frame = JSON.parse(event.data) as RuntimeFrame;
    63	          if (!frame || typeof frame.type !== "string") return;
    64	          setLastFrame(frame);
