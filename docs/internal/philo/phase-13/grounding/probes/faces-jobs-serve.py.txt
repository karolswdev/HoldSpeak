"""PHILO-13 grounding, faces half, Job A: one real hub on a throwaway HOME,
populated with a working week for a Senior Software Architect with three
reports, kept alive for the Tuesday-spine walks.

Run from the repository root, with HOME set to a fresh mktemp dir:

    H=$(mktemp -d); trap 'rm -rf "$H"' EXIT INT TERM
    HOME=$H PLAYWRIGHT_BROWSERS_PATH=/Users/karol/Library/Caches/ms-playwright \
      npm_config_cache=/Users/karol/.npm \
      uv run python docs/internal/philo/phase-13/grounding/probes/faces-jobs-serve.py.txt

It writes $HOME/hub.json ({url, token, ids}) and serves until $HOME/stop exists.

Every record comes from a real producer: the HTTP routes for projects,
updates, decisions, notes, people (encrypted sidecar on a FILE key in this
HOME), the folder destination, the calendar source (an ICS file in this HOME
through PUT /api/settings and the real ingest conductor) and the brief
(POST /api/brief/generate). Meetings with summaries and their decision
artifact go through the real meeting repository (db.meetings.save_meeting,
db.plugins.record_artifact + backfill_decisions), as the Phase 11 glass rigs
mint them: no route creates a summarized meeting while the engine is off.
The Ask Keep artifact goes through AskService.keep (no model runs).

No engine is assigned (the engine is OFF for this walk). The hub has no
recorder: POST /api/meeting/start cannot open a microphone.
"""

import os
import sys
import time
import urllib.request
from datetime import datetime, timedelta, timezone
from pathlib import Path

ROOT = Path.cwd()

sys.path.insert(0, str(ROOT))

def api(method: str, path: str, body: dict | None = None) -> dict:
    req = urllib.request.Request(
        base + path, method=method,
        data=json.dumps(body).encode() if body is not None else None,
        headers={"authorization": f"Bearer {TOKEN}", "content-type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=60) as resp:
        raw = resp.read().decode()
        return json.loads(raw) if raw.strip().startswith(("{", "[")) else {"_text": raw}


from holdspeak.db import get_database  # noqa: E402
from holdspeak.db.decisions import backfill_decisions  # noqa: E402
from holdspeak.meeting_session import IntelSnapshot, MeetingState, TranscriptSegment  # noqa: E402

db = get_database()
assert str(Path(db.db_path).resolve()).startswith(str(HOME)), db.db_path
ids: dict = {"home": str(HOME), "db": str(db.db_path)}

api("PUT", "/api/setup/onboarding", {"disposition": "completed"})

# ── projects + published updates (the Room's own routes) ──
projects = {}
for name, body in (
    ("Payments ledger cutover", "# Payments ledger cutover\n\n- Dry run 2 passed on staging.\n- Freeze date proposed: Nov 5.\n- Risk: reconciliation lag on the EU shard.\n"),
    ("Platform API v3", "# Platform API v3\n\n- Pagination contract agreed.\n- OpenTelemetry adopted for traces.\n- Next: SDK beta for two internal teams.\n"),
    ("Observability uplift", "# Observability uplift\n\n- Dashboards for checkout are live.\n- Alert noise down 40%.\n"),

    pid = api("POST", "/api/projects", {"name": name})["project"]["id"]
    up = api("POST", f"/api/projects/{pid}/updates/draft", {})["update"]["id"]
    api("PUT", f"/api/updates/{up}", {"body_md": body})
    api("POST", f"/api/updates/{up}/publish", {})

# ── meetings this week, through the real repository ──
now = datetime.now()
def ago(days: float, hour: int) -> datetime:
    d = (now - timedelta(days=days)).replace(hour=hour, minute=0, second=0, microsecond=0)
    return d if d < now else now - timedelta(hours=1)

def item(mid: str, i: int, task: str, owner: str) -> dict:
    return {"id": f"{mid}-a{i}", "task": task, "owner": owner, "due": None, "status": "pending",
            "review_state": "accepted", "source_timestamp": None, "created_at": now.isoformat(timespec="seconds")}

meetings = [
    ("p13-cutover-sync", "Ledger cutover sync", ago(2, 10), ["Ledger cutover", "Reconciliation"],
     "We agreed to freeze the old ledger on Nov 5. Reconciliation lag on the EU shard is the top risk. Priya owns the dry-run report.",
     [("Send the dry-run report", "Priya"), ("Size the EU shard backfill", "Marcus")], "Payments ledger cutover"),
    ("p13-api-review", "API v3 design review", ago(1, 14), ["Pagination", "Tracing"],
     "Cursor pagination is the contract. OpenTelemetry is the trace standard. Dana drafts the SDK beta plan.",
     [("Draft the SDK beta plan", "Dana")], "Platform API v3"),
    ("p13-incident", "Checkout latency review", ago(0, 8), ["Incident", "Checkout"],
     "Latency came from a cold cache after deploy. Add a warm-up step. Marcus owns the runbook change.",
     [("Add cache warm-up to the deploy runbook", "Marcus")], "Observability uplift"),
]
for mid, title, started, topics, summary, actions, proj in meetings:

        segments=[TranscriptSegment(text=summary, speaker="Karol", start_time=1.0, end_time=9.0)],
        intel=IntelSnapshot(timestamp=1.0, topics=topics, summary=summary,
                            action_items=[item(mid, i, t, o) for i, (t, o) in enumerate(actions)]),
    ))
    api("POST", f"/api/projects/{projects[proj]['id']}/meetings/{mid}", {})
# one meeting with no summary (recorded, engine off)
db.meetings.save_meeting(MeetingState(
    id="p13-vendor", started_at=ago(0, 9), ended_at=ago(0, 9) + timedelta(minutes=30), title="Vendor call: card network",
    segments=[TranscriptSegment(text="We discussed the card network SLA.", speaker="Karol", start_time=1.0, end_time=5.0)],

ids["meetings"] = [m[0] for m in meetings] + ["p13-vendor"]

# ── calendar: an ICS file in this HOME, today and tomorrow ──
utc = datetime.now(timezone.utc).replace(second=0, microsecond=0)
ics = HOME / "week.ics"
lines = ["BEGIN:VCALENDAR", "VERSION:2.0", "PRODID:-//HoldSpeak//PHILO13 jobs//EN"]
for uid, starts, mins, title in (
    ("p13-ev-1on1", utc + timedelta(hours=2), 30, "1:1 Priya / Karol"),
    ("p13-ev-sync", utc + timedelta(hours=4), 45, "Ledger cutover sync"),
    ("p13-ev-arch", utc + timedelta(hours=26), 60, "Architecture review"),
):
    lines += ["BEGIN:VEVENT", f"UID:{uid}", f"DTSTART:{starts.strftime('%Y%m%dT%H%M%SZ')}",
              f"DTEND:{(starts + timedelta(minutes=mins)).strftime('%Y%m%dT%H%M%SZ')}", f"SUMMARY:{title}", "END:VEVENT"]
lines += ["END:VCALENDAR", ""]
ics.write_text("\r\n".join(lines), encoding="utf-8")
api("PUT", "/api/settings", {"calendar": {"sources": [{"id": "p13-week", "label": "Work", "url": str(ics), "enabled": True}]}})
