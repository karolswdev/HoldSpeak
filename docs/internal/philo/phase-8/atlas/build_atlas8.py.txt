def line_of(path: str, symbol: str) -> int:
    for n, text in enumerate((ROOT / path).read_text().splitlines(), 1):
        if symbol in text:
            return n
    raise SystemExit(f"{path}: {symbol!r} not found")


def src(path: str, symbol: str, claim: str) -> dict:
    return {"path": path, "symbol": symbol, "line": line_of(path, symbol), "claim": claim}

def palette(query: str, at_width: int | None = None) -> list[dict]:

        {"kind": "ui", "action": "click", "adapter": "ui-pointer", "selector": "[aria-controls=desk-tool-shelf]"},
