caught = 0
for name, path, old, new, fences in MUTATIONS:
    file = Path(path)
    original = file.read_text()
    pairs = old if isinstance(old, list) else [(old, new)]
    mutated = original
    for before, after in pairs:
        assert mutated.count(before) == 1, (name, "the original text is not unique", before[:60])
        mutated = mutated.replace(before, after)
    try:
        file.write_text(mutated)
        run = subprocess.run([sys.executable, "-m", "pytest", "-q", "-x", "-p", "no:cacheprovider", *fences],
                             capture_output=True, text=True)
    finally:
        file.write_text(original)
    tail = [line for line in run.stdout.splitlines() if line.strip()][-1:]
    why = [line.strip()[:160] for line in run.stdout.splitlines() if line.startswith("E  ")][:1]
    red = run.returncode != 0
    caught += red
    print(f"{'CAUGHT' if red else 'MISSED'} {name}: rc={run.returncode} {tail[0] if tail else ''}")
    if why:
        print(f"    first assertion: {why[0]}")
print(f"{caught}/{len(MUTATIONS)} mutations caught")
sys.exit(0 if caught == len(MUTATIONS) else 1)
