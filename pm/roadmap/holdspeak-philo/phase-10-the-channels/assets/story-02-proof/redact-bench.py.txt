"""PHILO-10-02 GATE 1: the redactor's worst cases, timed one by one (run from the worktree root)."""
import random
import string
import time

from holdspeak.services.channel_contract import REDACT_SCAN_LIMIT, redact


def rnd(n: int, alphabet: str = string.ascii_letters + " ", seed: int = 7) -> str:
    rng = random.Random(seed)
    return "".join(rng.choice(alphabet) for _ in range(n))


error = rnd(2000, seed=11)
cases = {
    "Codex: 2000-char error x 10 MiB": (error, (rnd(1024 * 1024) * 10).encode()),
    "Codex: repetitive 10 MiB": ("a" * 1990 + "zzzzzzzzzz", (("a" * 9 + "b") * (1024 * 1024)).encode()),
    "2000-char error x 1 MiB (the scan limit)": (error, rnd(REDACT_SCAN_LIMIT).encode()),
}
for pattern in ("a" * 9 + "b", "ab", "aab", "a" * 30 + "b"):
    payload = (pattern * (REDACT_SCAN_LIMIT // len(pattern) + 1))[:REDACT_SCAN_LIMIT].encode()
    cases[f"near-miss {pattern[:5]}..({len(pattern)}) x 1 MiB"] = (rnd(2000, "ab", seed=11), payload)
worst = 0.0
for name, (text, payload) in cases.items():
    started = time.perf_counter()
    redact(text, payload)
    elapsed = time.perf_counter() - started
    worst = max(worst, elapsed)
    print(f"{elapsed:7.3f} s  {name}")
print(f"worst {worst:.3f} s (story 01 on Codex's cases: 8.68 s and 26.99 s; the fence bound: 2 s)")
