def run(args: list[str]) -> tuple[int, str]:
    with tempfile.TemporaryDirectory() as base:
        done = subprocess.run([sys.executable, "-m", "pytest", "-q", "-p", "no:cacheprovider", "-x",
                               f"--basetemp={base}/pt", TESTS, *args], capture_output=True, text=True)
    tail = [line for line in done.stdout.splitlines() if " passed" in line or " failed" in line]
    return done.returncode, tail[-1] if tail else done.stdout[-300:]


def main() -> int:
    code, tail = run([])
    print(f"BASELINE (no mutation): rc={code} {tail}")
    if code != 0:
        return 1
    missed = 0
    for label, path, old, new, args in MUTATIONS:
        target = Path(path)
        original = target.read_bytes()
        text = original.decode()
        pairs = list(zip(old, new)) if isinstance(old, tuple) else [(old, new)]
        if any(text.count(a) != 1 for a, _b in pairs):
            print(f"{label}: ANCHOR NOT FOUND")
            missed += 1
            continue
        for a, b in pairs:
            text = text.replace(a, b)
        try:
            target.write_text(text)
            code, tail = run(args)
        finally:
            target.write_bytes(original)
        red = code != 0
        missed += 0 if red else 1
        print(f"{'RED  ' if red else 'GREEN'} {label}: rc={code} {tail}")
    print(f"MUTATIONS: {len(MUTATIONS) - missed}/{len(MUTATIONS)} turned their fences red")
    return 0 if missed == 0 else 1
